Comanda			Descripció

$ mkdir carpeta/	crea una carpeta anomenada carpeta/

$ cd carpeta/		canvia el directori actual a carpeta/

$ cat fitxer		muestra el contenido del archivo

$ gedit fitxer &	edita un archivo y no bloquea la consola

$ ls			muestra todos los archivos y carpetas

$ rm fitxer		borra un archivo

$ git init		Permet inicialitzar un repositori git

$ git clone		clona un repositorio de git

$ git status		muestra el estado del repositorio

$ git add		agrega el archivo en el repositorio

$ git commit		confirma los cambios y adjunta un comentario

$ git push		sube los cambios hechos en el repositorio

$ git pull		descarga los cambios hechos en el repositorio
